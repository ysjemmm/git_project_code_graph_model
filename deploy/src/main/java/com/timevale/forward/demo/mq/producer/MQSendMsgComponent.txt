package com.timevale.mq_example.core.service.mq;

import com.timevale.framework.mq.client.producer.Msg;
import com.timevale.framework.mq.client.producer.OfferResult;
import com.timevale.framework.mq.client.producer.Producer;
import com.timevale.mq_example.core.service.mq.producer.MQTopicConfig;
import lombok.extern.log4j.Log4j2;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

/**
 * @author gwk_2
 * @date 2020/1/7 15:51
 */
@Service
@Log4j2
public class MQSendMsgComponent {

    @Autowired
    MQTopicConfig mqTopicConfig;
    @Autowired
    Producer producer;

    public void send(String str) {
        Msg msg = new Msg();
        msg.setTopic(mqTopicConfig.getTopic_1());
        //msg.setTags("tag 1"); // 设置 tag
        msg.setBody(str.getBytes());
        /*
         * 延迟消息;
         */
        //msg.setDelayMsgLevel(DelayMsgLevel.EIGHT); // 设置延迟消息;
        // 发送消息;
        OfferResult or = producer.offer(msg);
        if (or.isSuccess()) {
            log.info("send success");
            /*
             * TODO
             */
        } else {
            /*
            todo
             */
        }
    }

}

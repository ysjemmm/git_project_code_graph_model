package com.timevale.mq_example.core.service.mq.consumer;

import lombok.extern.log4j.Log4j2;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.annotation.PostConstruct;

/**
 * 消费者驱动类;
 *
 * @author gwk_2
 * @date 2020/1/7 15:27
 */
@Component
@Log4j2
public class MQConsumerConfig {

	
    /** 需要现在控制台创建 topic **/
    @Value("${topic1:topic1}")
    private String topic_1;
    /**
     * tag 表达式,默认就是 *;
     */
    @Value("${subExpression1:*}")
    private String subExpression_1;
    /**
     * 需要先在控制台创建 消费者 group;
     */
    @Value("${groupName1:groupName1}")
    private String groupName_1;

    /**
     * 需要现在控制台创建 topic;
     */
    @Value("${topic2:topic2}")
    private String topic_2;
    /**
     * tag 表达式,默认就是 *;
     */
    @Value("${subExpression2:*}")
    private String subExpression_2;
    /**
     * 需要先在控制台创建 消费者 group;
     */
    @Value("${groupName2:groupName2}")
    private String groupName_2;

    @Autowired
    private MQConsumerAdapter mqConsumerAdapter;
    @Autowired
    private Topic1Listener topic_1_listener;
    @Autowired
    private Topic2Listener topic_2_listener;

    @PostConstruct
    public void init() {
        /**
         * 同时启动多个消费者;
         */
        mqConsumerAdapter.startConsumer(topic_1, subExpression_1, groupName_1, topic_1_listener);
        mqConsumerAdapter.startConsumer(topic_2, subExpression_2, groupName_2, topic_2_listener);
        log.info("消费者启动完毕.");
    }
}
